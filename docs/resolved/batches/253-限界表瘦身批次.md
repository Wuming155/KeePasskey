```python
SEC = re.compile(r"§[0-9]+(?:\.[0-9]+)*")
TICK = re.compile(r"`([^`\n]+)`")
NUM = re.compile(
    r"0x[0-9A-Fa-f]+"
    r"|[0-9]+(?:\.[0-9]+)?\s*[×x]\s*10\^?[-0-9]+"

    r"|2\^[0-9]+"
)
CAPS = re.compile(r"\b[A-Z][A-Z0-9]*(?:_[A-Z0-9]+)+\b")

def tokens(text):
    ticks = set()
    for body in TICK.findall(text):
        if "/" in body or ".kt" in body or ".md" in body: ticks.add(body.strip())
        elif CAPS.fullmatch(body.strip()):              ticks.add(body.strip())

            "数字常量": {re.sub(r"\s+", "", m) for m in NUM.findall(text)}}

# 逐组打印 MISSING / ADDED，退出码 = MISSING 总数
```
